// controllers/contactController.js
const { body, validationResult } = require('express-validator');
const Contact = require('../models/Contact');

exports.submitContactForm = [
  // Validation rules
  body('name').notEmpty().withMessage('Name is required'),
  body('email').isEmail().withMessage('A valid email is required'),
  body('message').isLength({ min: 10 }).withMessage('Message must be at least 10 characters'),

  async (req, res) => {
    // Check validation results
    const errors = validationResult(req);
    if (!errors.isEmpty()) {
      return res.status(400).json({ errors: errors.array() });
    }

    try {
      // Extract form data
      const { name, email, message } = req.body;

      // Save to database
      const contactEntry = new Contact({ name, email, message });
      await contactEntry.save();

      // Respond with success
      res.status(201).json({ message: 'Form submitted successfully!' });
    } catch (err) {
      res.status(500).json({ error: 'Server error while processing form.' });
    }
  }
];

AI Coach: Thanks for sharing that snippet, Martin! Looking at your submitContactForm function, how are you extracting the user's input from the incoming request, and what role does req.body play in making that data accessible to your database logic?

Martin: When I handle a form submission like in my submitContactForm function, I’m pulling the user’s input straight out of req.body. That’s the object Express builds for me after parsing the incoming request body with middleware like express.json() or urlencoded(). So when a client posts data—say, name, email, and message—those fields arrive in the request body. Express parses them and makes them accessible as properties of req.body. That’s why I can write: const { name, email, message } = req.body; At that point, I’ve got the user’s input in clean variables I can validate, transform, or push into my database. In other words, req.body is the bridge between the raw HTTP payload and the structured data my logic depends on. Without it, I’d be stuck parsing the raw request stream myself. For me, the role of req.body is crucial—it’s what makes the user’s submitted data usable in my controller logic and ensures I can store it reliably in my database.

});

  body('username')

    .isLength({ min: 3 }).withMessage('Username must be at least 3 characters'),
  body('email')
    .isEmail().withMessage('Please enter a valid email'),
  body('password')
    .isLength({ min: 6 }).withMessage('Password must be at least 6 characters'),

  async (req, res) => {
    const errors = validationResult(req);
    if (!errors.isEmpty()) {
      // Send back validation errors
      return res.status(400).json({ errors: errors.array() });
    }

    try {
      // If validation passes, proceed with database logic
      const { username, email, password } = req.body;
      const user = new User({ username, email, password });
      await user.save();

      res.status(201).json({ message: 'User created successfully!' });
    } catch (err) {
      res.status(500).json({ error: 'Server error while creating user.' });
    }
  }
];

Why server‑side validation is essential:
Client‑side checks can be bypassed: A malicious user could disable JavaScript or craft a custom request that ignores HTML validation.
Data integrity: The server ensures that only properly formatted, safe data enters the database.
Security: Prevents injection attacks, malformed inputs, or unexpected values that could break logic.
Consistency: Guarantees that all clients (even those without modern browsers) are held to the same rules.
For me, server‑side validation is the last line of defense—it makes sure that no matter how the data arrives, my application enforces its rules before trusting or storing anything.
